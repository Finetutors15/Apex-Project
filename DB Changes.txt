--------STUDENT_FEES_HISTORY------------
REG_FEE_AMT     NUMBER;

PAST_PAPER_FEE_AMT NUMBER;
MONTH_FEE_DISC_AMT  NUMBER;
---------------------------------
create or replace TRIGGER TRG_STUDENT_FEES_HISTORY
BEFORE  UPDATE OR DELETE ON STUDENT_FEES
FOR EACH ROW
DECLARE
    v_action_type VARCHAR2(10);  
BEGIN

    IF UPDATING THEN
        v_action_type := 'UPDATE';
    ELSIF DELETING THEN
        v_action_type := 'DELETE';
    END IF;
    -- Insert old fee record into history table
    INSERT INTO STUDENT_FEES_HISTORY (
        HISTORY_ID, FEE_ID, STUDENT_ID, SITE_ID, REGISTRATION_FEE, BOOK_FEE, EXAM_FEE, 
        HOURLY_FEE, PAST_PAPER_FEE,TOTAL_FEE,FEE_START_DATE, DUE_DATE, PAY_ID, 
        REG_FEE_AMT, BOOK_FEE_AMT, PAST_PAPER_FEE_AMT, MONTH_FEE_DISC_AMT,
        ACTION_TYPE,FEED_BY, FEED_DATE
    ) VALUES (
        SEQ_STUDENT_FEES_HISTORY.NEXTVAL, :OLD.FEE_ID, :OLD.STUDENT_ID, :OLD.SITE_ID, 
        :OLD.REGISTRATION_FEE, :OLD.BOOK_FEE, :OLD.EXAM_FEE, 
        :OLD.HOURLY_FEE, :OLD.PAST_PAPER_FEE, :OLD.TOTAL_FEE,:OLD.FEE_START_DATE,
        :OLD.DUE_DATE, :OLD.PAY_ID, 
        :OLD.REG_FEE_AMT, :OLD.BOOK_FEE_AMT, :OLD.PAST_PAPER_FEE_AMT, :OLD.MONTH_FEE_DISC_AMT,
        v_action_type, NVL(V('APP_USER'), 'SYSTEM'), SYSDATE
    );

END;
