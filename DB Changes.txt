create or replace TRIGGER trg_generate_student_uk_id
BEFORE INSERT ON STUDENTS
FOR EACH ROW
DECLARE
    V_SITE_ID NUMBER;
    V_SITE_NAME VARCHAR2(250);
BEGIN
    SELECT COUNT(*) + 1
    INTO V_SITE_ID
    FROM STUDENTS
    ;
    SELECT SITE_NAME INTO V_SITE_NAME
      FROM SITES S
     WHERE S.SITE_ID = :NEW.SITE_ID
     ; 
    --WHERE STUDENT_ID = :NEW.STUDENT_ID;

    :NEW.STUDENT_UK_ID := V_SITE_NAME ||'-'||:NEW.STUDENT_ID || '-' || :NEW.SITE_ID;
END;
/
